```python
from django.http import JsonResponse
import time
import jwt
import os
from ratelimit.decorators import ratelimit

class APIGatewayMiddleware:
    def __init__(self, get_response):
        self.get_response = get_response
        self.api_keys = set(os.environ.get('API_KEYS', '').split(','))
        self.jwt_secret = os.environ.get('JWT_SECRET', 'your-secret-key')

    def __call__(self, request):
        # 记录请求开始时间
        start_time = time.time()

        # 1. 身份认证
        auth_header = request.headers.get('Authorization', '')
        if auth_header.startswith('Bearer '):
            token = auth_header[7:]
            try:
                decoded = jwt.decode(token, self.jwt_secret, algorithms=['HS256'])
                request.user_info = decoded
            except jwt.InvalidTokenError:
                return JsonResponse({'error': 'Invalid token'}, status=401)
        elif 'X-API-Key' in request.headers:
            api_key = request.headers['X-API-Key']
            if api_key not in self.api_keys:
                return JsonResponse({'error': 'Invalid API key'}, status=401)
            request.api_key = api_key
        # 公开API不需要认证
        elif not self._is_public_api(request.path):
            return JsonResponse({'error': 'Authentication required'}, status=401)

        # 2. 请求路由已由Django URL配置处理

        # 3. 限流保护已通过装饰器实现

        # 处理请求
        response = self.get_response(request)

        # 记录请求处理时间
        processing_time = time.time() - start_time
        response['X-Processing-Time'] = str(processing_time)

        return response

    def _is_public_api(self, path):
        public_paths = ['/api/health/', '/api/chat/']
        for public_path in public_paths:
            if path.startswith(public_path):
                return True
        return False

# 视图装饰器示例
def api_view_with_rate_limit(view_func):
    @ratelimit(key='user', rate='100/h')  # 限制每小时100次请求
    @ratelimit(key='ip', rate='50/h')    # 限制每小时50次IP请求
    def _wrapped_view(request, *args, **kwargs):
        was_limited = getattr(request, 'limited', False)
        if was_limited:
            return JsonResponse({'error': 'Rate limit exceeded'}, status=429)
        return view_func(request, *args, **kwargs)
    return _wrapped_view
```

```python
# orchestrator/scheduler.py
from typing import List, Dict, Any, Optional
from concurrent.futures import ThreadPoolExecutor
import uuid
import logging

logger = logging.getLogger(__name__)

class AgentScheduler:
    def __init__(self, max_workers: int = 4):
        self.executor = ThreadPoolExecutor(max_workers=max_workers)
        self.running_tasks = {}
        self.memory_manager = MemoryManager()

    def schedule_task(self, pipeline: List[str], params: Dict[str, Any]) -> str:
        """调度新的智能体协作任务"""
        task_id = str(uuid.uuid4())
        logger.info(f'Scheduling task {task_id} with pipeline {pipeline}')
        
        # 提交任务到线程池
        future = self.executor.submit(self._execute_pipeline, task_id, pipeline, params)
        
        # 保存任务信息
        self.running_tasks[task_id] = {
            'future': future,
            'status': 'running',
            'pipeline': pipeline,
            'params': params,
            'result': None
        }
        
        return task_id

    def get_task_status(self, task_id: str) -> Dict[str, Any]:
        """获取任务状态"""
        if task_id not in self.running_tasks:
            return {'status': 'not_found'}
        
        task = self.running_tasks[task_id]
        if task['future'].done():
            try:
                result = task['future'].result()
                task['status'] = 'completed'
                task['result'] = result
            except Exception as e:
                task['status'] = 'failed'
                task['error'] = str(e)
                logger.error(f'Task {task_id} failed: {str(e)}')
        
        return {
            'status': task['status'],
            'pipeline': task['pipeline'],
            'result': task.get('result'),
            'error': task.get('error')
        }

    def _execute_pipeline(self, task_id: str, pipeline: List[str], params: Dict[str, Any]) -> Dict[str, Any]:
        """执行智能体协作流水线"""
        from agents.factory import AgentFactory
        from agents.base import AgentResult
        
        agent_factory = AgentFactory()
        context = {'task_id': task_id}
        results = {}
        
        try:
            # 加载任务记忆
            task_memory = self.memory_manager.load_task_memory(task_id)
            if task_memory:
                context.update(task_memory)
            
            # 执行流水线中的每个智能体
            for agent_name in pipeline:
                logger.info(f'Task {task_id}: Executing agent {agent_name}')
                
                # 创建智能体实例
                agent = agent_factory.create_agent(agent_name)
                agent_params = params.get(agent_name, {})
                
                # 执行智能体
                result = agent.run(agent_params, context)
                results[agent_name] = result.to_dict()
                
                if result.success:
                    # 更新上下文
                    context.update(result.data)
                    # 保存中间结果到记忆
                    self.memory_manager.save_intermediate_result(task_id, agent_name, result.data)
                else:
                    raise Exception(f'{agent_name} failed: {result.error}')
            
            # 任务完成，保存最终结果
            self.memory_manager.save_final_result(task_id, context)
            
            return {
                'success': True,
                'results': results,
                'context': context
            }
        except Exception as e:
            # 保存错误信息
            self.memory_manager.save_error(task_id, str(e))
            raise

class MemoryManager:
    """管理智能体协作的记忆"""
    
    def __init__(self):
        self.task_memories = {}
        
    def load_task_memory(self, task_id: str) -> Optional[Dict[str, Any]]:
        """加载任务记忆"""
        return self.task_memories.get(task_id, {})
        
    def save_intermediate_result(self, task_id: str, agent_name: str, result: Dict[str, Any]):
        """保存中间结果"""
        if task_id not in self.task_memories:
            self.task_memories[task_id] = {}
        self.task_memories[task_id][f'{agent_name}_result'] = result
        
    def save_final_result(self, task_id: str, context: Dict[str, Any]):
        """保存最终结果"""
        self.task_memories[task_id]['final_context'] = context
        
    def save_error(self, task_id: str, error: str):
        """保存错误信息"""
        if task_id not in self.task_memories:
            self.task_memories[task_id] = {}
        self.task_memories[task_id]['error'] = error

# 智能体工厂类
class AgentFactory:
    def __init__(self):
        self._agents = {}
    
    def create_agent(self, agent_name: str):
        """创建智能体实例"""
        if agent_name not in self._agents:
            # 动态导入智能体类
            if agent_name == 'DataAgent':
                from agents.data_agent import DataAgent
                self._agents[agent_name] = DataAgent()
            elif agent_name == 'AnalysisAgent':
                from agents.analysis_agent import AnalysisAgent
                self._agents[agent_name] = AnalysisAgent()
            elif agent_name == 'WritingAgent':
                from agents.writing_agent import WritingAgent
                self._agents[agent_name] = WritingAgent()
            elif agent_name == 'VideoAgent':
                from agents.video_agent import VideoAgent
                self._agents[agent_name] = VideoAgent()
            else:
                raise ValueError(f'Unknown agent: {agent_name}')
        
        return self._agents[agent_name]
```

```python
# orchestrator.py
from typing import List, Dict, Any
from .agents.base import AgentResult
from .agents.data_agent import DataAgent
from .agents.analysis_agent import AnalysisAgent
from .agents.writing_agent import WritingAgent
from .agents.video_agent import VideoAgent

class AgentOrchestrator:
    def __init__(self):
        self.agents = {
            "DataAgent": DataAgent(),
            "AnalysisAgent": AnalysisAgent(),
            "WritingAgent": WritingAgent(),
            "VideoAgent": VideoAgent()
        }
    
    def run_pipeline(self, ticker: str, pipeline: List[str], params: Dict[str, Any] = None) -> Dict[str, Any]:
        """运行智能体协作流水线"""
        context = {"ticker": ticker}
        results = {}
        
        for agent_name in pipeline:
            if agent_name in self.agents:
                agent = self.agents[agent_name]
                agent_params = params.get(agent_name, {}) if params else {}
                
                result = agent.run(agent_params, context)
                results[agent_name] = result
                
                if result.success:
                    context[agent_name] = result
                    # 将当前智能体的结果合并到上下文中
                    context.update(result.data)
                else:
                    # 处理失败情况
                    return {
                        "success": False,
                        "error": f"{agent_name} failed: {result.error}",
                        "results": results
                    }
            else:
                return {
                    "success": False,
                    "error": f"Unknown agent: {agent_name}",
                    "results": results
                }
        
        return {
            "success": True,
            "results": results,
            "context": context
        }


def run_plan(plan: List[Dict[str, Any]], context: Dict[str, Any] = None) -> Dict[str, Any]:
    """执行计划"""
    orchestrator = AgentOrchestrator()
    results = []
    
    for step in plan:
        pipeline = step.get("pipeline", [])
        params = step.get("params", {})
        ticker = step.get("ticker") or context.get("ticker")
        
        if not ticker:
            return {"success": False, "error": "Missing ticker"}
        
        result = orchestrator.run_pipeline(ticker, pipeline, params)
        results.append(result)
        
        if not result["success"]:
            return {"success": False, "error": result["error"], "results": results}
    
    return {"success": True, "results": results}
```

```python
from typing import List, Dict, Any
import requests
import json
from datetime import datetime, timedelta

class MarketProviderError(Exception):
    pass


def _parse_window(window: str) -> tuple:
    """解析时间窗口"""
    # 支持格式：1D, 1W, 1M, 3M, 6M, 1Y
    units = {
        'D': 1,
        'W': 7,
        'M': 30,
        'Y': 365
    }
    
    try:
        number = int(window[:-1])
        unit = window[-1].upper()
        days = number * units.get(unit, 1)
        
        end_date = datetime.now()
        start_date = end_date - timedelta(days=days)
        
        return start_date.strftime('%Y%m%d'), end_date.strftime('%Y%m%d')
    except (ValueError, IndexError):
        raise MarketProviderError(f"Invalid window format: {window}")


def fetch_kline(ticker: str, window: str = "3M") -> List[Dict[str, Any]]:
    """获取股票K线数据"""
    try:
        # 这里使用模拟数据，实际实现应该调用真实的市场数据API
        # 例如 Yahoo Finance API, Alpha Vantage 等
        
        start_date, end_date = _parse_window(window)
        
        # 模拟数据生成逻辑
        import random
        dates = []
        current = datetime.strptime(start_date, '%Y%m%d')
        end = datetime.strptime(end_date, '%Y%m%d')
        
        while current <= end:
            if current.weekday() < 5:  # 只包含工作日
                dates.append(current.strftime('%Y-%m-%d'))
            current += timedelta(days=1)
        
        # 生成随机价格数据
        base_price = random.uniform(50, 200)
        prices = []
        
        for date in dates:
            change = random.uniform(-2, 2)
            close = max(1, base_price + change)
            open_price = close + random.uniform(-1, 1)
            high = max(open_price, close) + random.uniform(0, 1)
            low = min(open_price, close) - random.uniform(0, 1)
            volume = random.randint(1000000, 10000000)
            
            prices.append({
                "date": date,
                "open": round(open_price, 2),
                "high": round(high, 2),
                "low": round(low, 2),
                "close": round(close, 2),
                "volume": volume
            })
            
            base_price = close
        
        return prices
    except Exception as e:
        raise MarketProviderError(f"Failed to fetch kline data: {str(e)}")
```

```python
class ABTestManager:
```
